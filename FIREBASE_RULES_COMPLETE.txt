rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // --------------------------
    // Conversations
    // --------------------------
    match /conversations/{conversationId} {
      allow read, write: if request.auth != null 
        && resource.data.participants.hasAny([request.auth.uid]);

      allow create: if request.auth != null 
        && request.auth.uid in request.data.participants;
    }

    match /conversations/{conversationId}/messages/{messageId} {
      allow read, write: if request.auth != null 
        && get(/databases/$(database)/documents/conversations/$(conversationId))
          .data.participants.hasAny([request.auth.uid]);
    }

    // --------------------------
    // LEGACY MESSAGES (fixed)
    // --------------------------
    match /messages/{messageId} {
      // Only allow a user to write their own message
      allow create: if request.auth != null
        && request.auth.uid == request.data.userId;

      allow read, update, delete: if request.auth != null
        && request.auth.uid == resource.data.userId;
    }

    // --------------------------
    // Notifications
    // --------------------------
    match /notifications/{notificationId} {
      // Users can read notifications sent to them (recipientId) or sent by them
      // Also support legacy userId field for backward compatibility
      allow read: if request.auth != null 
        && (resource.data.recipientId == request.auth.uid 
            || resource.data.senderId == request.auth.uid
            || resource.data.userId == request.auth.uid);

      // Authenticated users can create notifications
      // This allows anyone authenticated to create notifications (for booking requests, messages, etc.)
      allow create: if request.auth != null;

      // Users can update their own notifications (mark as read)
      allow update: if request.auth != null
        && (resource.data.recipientId == request.auth.uid
            || resource.data.userId == request.auth.uid);

      // Users can delete their own notifications
      allow delete: if request.auth != null
        && (resource.data.recipientId == request.auth.uid
            || resource.data.userId == request.auth.uid);
    }

    // --------------------------
    // Reviews
    // --------------------------
    match /reviews/{reviewId} {
      // Anyone can read reviews
      allow read: if true;
      
      // Users can create reviews if authenticated and userId matches
      allow create: if request.auth != null
        && request.auth.uid == request.data.userId
        && request.data.rating >= 1
        && request.data.rating <= 5
        && request.data.comment.size() >= 10
        && request.data.driverId != null;
      
      // Users can update reviews:
      // 1. Owner can update their own review content (rating, comment, etc.)
      // 2. Anyone can update likes/dislikes (for like/dislike functionality)
      allow update: if request.auth != null && (
        // Owner updating their own review
        (request.auth.uid == resource.data.userId
          && request.data.rating >= 1
          && request.data.rating <= 5
          && request.data.comment.size() >= 10)
        ||
        // Like/dislike updates - allow if only updating interaction fields
        (request.data.diff(resource.data).affectedKeys()
          .hasOnly(['likes', 'dislikes', 'likedBy', 'dislikedBy', 'lastUpdated']))
      );
      
      // Users can delete their own reviews
      allow delete: if request.auth != null
        && request.auth.uid == resource.data.userId;
    }

    // --------------------------
    // Reported Reviews
    // --------------------------
    match /reportedReviews/{reportId} {
      // Only admins can read reports (for now, allow authenticated users to create)
      allow read: if request.auth != null;
      
      // Authenticated users can report reviews
      allow create: if request.auth != null
        && request.auth.uid == request.data.reporterId
        && request.data.reason.size() >= 5
        && request.data.reviewId != null;
    }

    // --------------------------
    // Bookings (secure create) - FIXED
    // --------------------------
    match /bookings/{bookingId} {
      // Customers can create bookings
      // Check: authenticated, customerId matches auth, required fields exist
      allow create: if request.auth != null
        && request.auth.uid == request.data.customerId
        && request.data.driverId != null
        && request.data.selectedDates != null
        && request.data.totalPrice != null;

      // Customers can read their own bookings
      // Drivers can read bookings for their services
      allow read: if request.auth != null
        && (request.auth.uid == resource.data.customerId
            || request.auth.uid == resource.data.driverId);

      // Drivers can update bookings (accept/cancel)
      // Customers can update their own bookings (cancel) - limited fields
      allow update: if request.auth != null
        && (request.auth.uid == resource.data.driverId
            || request.auth.uid == resource.data.customerId);

      // Only customers can delete their own bookings
      allow delete: if request.auth != null
        && request.auth.uid == resource.data.customerId;
    }

    // --------------------------
    // Tourists (profile)
    // --------------------------
    match /tourists/{touristId} {
      allow read, write: if request.auth != null
        && request.auth.uid == touristId;
    }

    // --------------------------
    // Service Providers (fixed)
    // --------------------------
    match /serviceProviders/{providerId} {
      allow read: if true;

      allow create: if request.auth != null
        && request.auth.uid == providerId;

      allow update, delete: if request.auth != null
        && request.auth.uid == providerId;
    }

    // --------------------------
    // Online Status
    // --------------------------
    match /onlineStatus/{userId} {
      allow read: if request.auth != null;

      allow write: if request.auth != null
        && request.auth.uid == userId;
    }
  }
}
